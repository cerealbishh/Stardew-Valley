async function fetchBridge(){

  clearTimeout(S.retry);

  const ip = $('bridge-ip').value.trim();

  if(!ip){ setBridgeOff(); return; }

  localStorage.setItem('sdv_ip', ip);

  const saveUrl = ip.startsWith('http') ? ip.replace(/\/+$/,'')+'/save' : `https://${ip}:${BRIDGE_PORT}/save`;

  try{

    const res = await fetch(saveUrl,{signal:AbortSignal.timeout(5000)});

    if(!res.ok) throw 0;

    const d = await res.json();

    if(d.error) throw 0;

    S.data=d; S.ok=true;

    setBridgeOn(d); renderFromData(d);

  }catch(_){ S.ok=false; setBridgeOff(); }

  S.retry = setTimeout(fetchBridge, RETRY_MS);

}

function setBridgeOff(){

}

function spawnJunimo(){

  const stage = $('junimo-stage'); if(!stage) return;

  const cols = JUNIMO_COLS[season()];

  const col  = cols[Math.floor(Math.random()*cols.length)];

  const j = document.createElement('div');

  j.className = 'junimo';

  j.innerHTML = jSVG(col);

  j.style.cssText = `left:${5+Math.random()*80}%;bottom:0;animation-duration:${0.5+Math.random()*0.7}s`;

  stage.appendChild(j);

  setTimeout(()=>{ j.style.opacity='0'; setTimeout(()=>j.remove(),450); }, 2000+Math.random()*1800);

}


function startJunimos(){

  spawnJunimo();

  setInterval(()=>{ if(Math.random()>0.2) spawnJunimo(); }, 1800);

}

let pInt;

function spawnParticles(s){

  clearInterval(pInt);

  document.querySelectorAll('.particle').forEach(p=>p.remove());

  const emojis = PARTICLES[s]||PARTICLES.spring;

  function one(){

    const p = document.createElement('div');

    p.className='particle';

    p.textContent = emojis[Math.floor(Math.random()*emojis.length)];

    const dur = 9+Math.random()*9;

    p.style.cssText=`left:${Math.random()*92}%;bottom:-4%;font-size:${13+Math.random()*11}px;animation-duration:${dur}s`;

    document.body.appendChild(p);

    setTimeout(()=>p.remove(), dur*1000+500);

  }

  for(let i=0;i<5;i++) setTimeout(one, i*700);

  pInt = setInterval(one, 2800);

}
